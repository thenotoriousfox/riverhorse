add_library(psx_gpu riverhorse_psx_gpu.c plugins/gpulib/prim.c)
target_include_directories(psx_gpu PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_include_directories(psx_gpu PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/win32")
endif()
target_compile_definitions(psx_gpu PRIVATE GPU_NEON TEXTURE_CACHE_4BPP TEXTURE_CACHE_8BPP)

  enable_language(ASM)
  target_sources(psx_gpu PRIVATE plugins/gpu_neon/psx_gpu/psx_gpu_arm_neon.S)
  target_include_directories(psx_gpu PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_compile_definitions(psx_gpu PRIVATE NEON_BUILD)
elseif(CPU_ARCH_X64 OR CPU_ARCH_ARM64)
  target_sources(psx_gpu PRIVATE plugins/gpu_neon/psx_gpu/psx_gpu_simd.c)
  target_compile_definitions(psx_gpu PRIVATE NEON_BUILD SIMD_BUILD)
endif()
